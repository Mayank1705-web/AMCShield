#
# IMPORTANT:
#
# This middleware does NOT access request.session.
#
# CSRF is handled by the dedicated /api/auth/csrf endpoint
# and route-level validation in the authentication/security
# layer.
#
# Keeping session-dependent CSRF logic out of this middleware
# eliminates the SessionMiddleware ordering problem permanently.

@app.middleware("http")
async def security_middleware(request: Request, call_next):
    response = await call_next(request)

    # --------------------------------------------------------
    # Browser security headers
    # --------------------------------------------------------

    response.headers["X-Content-Type-Options"] = "nosniff"

    response.headers["X-Frame-Options"] = "DENY"

    response.headers["Referrer-Policy"] = (
        "strict-origin-when-cross-origin"
    )

    response.headers["Permissions-Policy"] = (
        "camera=(), "
        "microphone=(), "
        "geolocation=()"
    )

    response.headers["Content-Security-Policy"] = (
        "default-src 'self'; "
        "script-src 'self'; "
        "style-src 'self' https://fonts.googleapis.com; "
        "font-src 'self' https://fonts.gstatic.com data:; "
        "img-src 'self' data: blob:; "
        "connect-src 'self'; "
        "object-src 'none'; "
        "base-uri 'self'; "
        "form-action 'self'; "
        "frame-ancestors 'none'; "
        "manifest-src 'self'; "
        "worker-src 'self' blob:;"
    )

    # --------------------------------------------------------
    # HSTS
    # --------------------------------------------------------

    if settings.session_https_only:
        response.headers["Strict-Transport-Security"] = (
            "max-age=31536000; includeSubDomains"
        )

    # --------------------------------------------------------
    # API cache control
    # --------------------------------------------------------

    if request.url.path.startswith(
        f"{settings.api_prefix}/"
    ) or request.url.path.endswith(".html") or request.url.path in {"/", "/admin"}:
        response.headers["Cache-Control"] = "no-store, max-age=0"
        response.headers["Pragma"] = "no-cache"

    return response

def _register_admin_asset(
    url_path: str,
    filename: str,
    media_type: str | None,
    is_page: bool,
):
    @app.get(url_path, include_in_schema=False)
    def _serve(
        request: Request,
        db: Session = Depends(get_db),
    ):
        if is_admin_session(request, db):
            return FileResponse(
                FRONTEND_DIR / filename,
                media_type=media_type,
                headers=NO_STORE,
            )

        if not is_page:
            return Response(
                status_code=404,
                headers=NO_STORE,
            )

        if request.session.get("user"):
            return RedirectResponse(
                "/dashboard.html",
                status_code=302,
                headers=NO_STORE,
            )

        return RedirectResponse(
            "/",
            status_code=302,
            headers=NO_STORE,
        )


_register_admin_asset("/admin.html", "admin.html", "text/html", True)

async function fetchAPI(endpoint, options = {}) {
  const response = await fetch(`${API_BASE_URL}${endpoint}`, {
    ...options,
    headers: {
      "Accept": "application/json",
      ...(options.headers || {})
    }
  });

  if (!response.ok) {
    let detail = "";
    try {
      const body = await response.json();
      detail = body.detail || body.message || "";
    } catch (_) {}
    throw new Error(detail || `API request failed (${response.status})`);
  }

  const body = await response.json();

  // Dashboard endpoints return { success: true, data: ... }.
  // Unwrap once so renderers receive the actual payload.
  if (
    body &&
    body.success === true &&
    Object.prototype.hasOwnProperty.call(body, "data")
  ) {
    return body.data;
  }

  return body;
}

function formatNumber(value) {
  if (typeof value !== "number" || !Number.isFinite(value)) {
    return "N/A";
  }

  return new Intl.NumberFormat("en-IN", {
    notation: Math.abs(value) >= 100000 ? "compact" : "standard",
    maximumFractionDigits: 1
  }).format(value);
}

function formatPercent(value) {
  if (typeof value !== "number" || !Number.isFinite(value)) {
    return "N/A";
  }

  return `${value.toFixed(2)}%`;
}

function formatSNR(value) {
  if (typeof value !== "number" || !Number.isFinite(value)) {
    return "N/A";

  return `${value} dB`;
}

function normalizeSeries(payload, key) {
  if (!payload) return [];

  const rows = Array.isArray(payload) ? payload : payload.data;

  if (Array.isArray(rows)) {
    return rows
      .filter(row =>
        row &&
        row.snr != null &&
        row[key] != null &&
        Number.isFinite(Number(row.snr)) &&
        Number.isFinite(Number(row[key]))
      )
      .map(row => ({
        snr: Number(row.snr),
        value: Number(row[key]) * 100
      }));
  }

  const snr = payload.snr || payload.snr_levels;
  const values = payload[key];

  if (Array.isArray(snr) && Array.isArray(values)) {
    return snr
      .map((x, i) => ({
        snr: Number(x),
        value: Number(values[i]) * 100
      }))
      .filter(point =>
        Number.isFinite(point.snr) &&
        Number.isFinite(point.value)
      );
  }

  return [];
}

function drawLineChart(svg, series, options = {}) {
  svg.replaceChildren();

  const width = svg.clientWidth || 720;
  const height = svg.clientHeight || 290;

  const pad = {
    left: 44,
    right: 18,
    top: 18,
    bottom: 34
  };

  const w = width - pad.left - pad.right;
  const h = height - pad.top - pad.bottom;

  const all = series.flatMap(s => s.points);

  if (!all.length) return false;

  const xs = all.map(p => p.snr);
  const ys = all.map(p => p.value);

  const minX = Math.min(...xs);
  const maxX = Math.max(...xs);

  const minY = Math.min(0, Math.min(...ys));
  const maxY = Math.max(options.maxY ?? 100, Math.max(...ys));

  const xScale = x =>
    pad.left + ((x - minX) / ((maxX - minX) || 1)) * w;

  const yScale = y =>
    pad.top + h -
    ((y - minY) / ((maxY - minY) || 1)) * h;

  const grid = svgEl("g");
  const axis = svgEl("g");
  const paths = svgEl("g");

  const ticks = 5;

  for (let i = 0; i <= ticks; i++) {
    const y = minY + ((maxY - minY) * i / ticks);
    const yy = yScale(y);

    grid.appendChild(svgEl("line", {
      x1: pad.left,
      x2: width - pad.right,
      y1: yy,
      y2: yy,
      stroke: "rgba(210,225,216,.09)",
      "stroke-width": "1"
    }));

    const label = svgEl("text", {
      x: pad.left - 8,
      y: yy + 3,
      "text-anchor": "end",
      fill: "#77837c",
      "font-size": "9"
    });

    label.textContent = Number(y).toFixed(0);
    axis.appendChild(label);
  }

  const xTicks = [...new Set(xs)].sort((a, b) => a - b);
  const step = Math.max(1, Math.ceil(xTicks.length / 8));

  xTicks
    .filter((_, i) => i % step === 0 || i === xTicks.length - 1)
    .forEach(x => {
      const xx = xScale(x);

      const label = svgEl("text", {
        x: xx,
        y: height - 10,
        "text-anchor": "middle",
        fill: "#77837c",
        "font-size": "9"
      });

      label.textContent = x;
      axis.appendChild(label);
    });

  const colors = options.colors || [
    "#f4c542",
    "#20c9d8",
    "#52d273",
    "#ff7070",
    "#c58cff"
  ];

  series.forEach((s, idx) => {
    if (!s.points.length) return;

    const d = s.points
      .map((p, i) =>
        `${i ? "L" : "M"} ${xScale(p.snr).toFixed(2)} ${yScale(p.value).toFixed(2)}`
      )
      .join(" ");

    paths.appendChild(svgEl("path", {
      d,
      fill: "none",
      stroke: colors[idx % colors.length],
      "stroke-width": "2.2",
      "stroke-linecap": "round",
      "stroke-linejoin": "round"
    }));

    s.points.forEach(p => {
      const c = svgEl("circle", {
        cx: xScale(p.snr),
        cy: yScale(p.value),
        r: "2.7",
        fill: colors[idx % colors.length],
        stroke: "#07100d",
        "stroke-width": "1"
      });

      const title = svgEl("title");
      title.textContent =
        `${s.name} � SNR ${p.snr} dB � ${p.value.toFixed(2)}%`;

      c.appendChild(title);
      paths.appendChild(c);
    });
  });

  const xTitle = svgEl("text", {
    x: pad.left + w / 2,
    y: height + 1,
    "text-anchor": "middle",
    fill: "#77837c",
    "font-size": "9"
  });

  xTitle.textContent = "SNR (dB)";
  axis.appendChild(xTitle);

  svg.append(grid, paths, axis);

  return true;
}

function getAttackSeries(data, selected = "") {
  const rows = Array.isArray(data) ? data : data?.data;

  if (Array.isArray(rows) && rows.length && rows[0]?.attack) {
    const methods = [
      ...new Set(rows.map(row => row.attack).filter(Boolean))
    ];

    const series = [];

    methods
      .filter(method => !selected || method === selected)
      .forEach(method => {
        const methodRows = rows.filter(
          row => row.attack === method
        );

        const baselinePoints = methodRows
          .filter(row =>
            row.snr != null &&
            row.baseline_asr != null
          )
          .map(row => ({
            snr: Number(row.snr),
            value: Number(row.baseline_asr) * 100
          }))
          .filter(point =>
            Number.isFinite(point.snr) &&
            Number.isFinite(point.value)
          );

        const robustPoints = methodRows
          .filter(row =>
            row.snr != null &&
            row.robust_asr != null
          )
          .map(row => ({
            snr: Number(row.snr),
            value: Number(row.robust_asr) * 100
          }))
          .filter(point =>
            Number.isFinite(point.snr) &&
            Number.isFinite(point.value)
          );

        if (baselinePoints.length) {
          series.push({
            name: `${method} � Baseline`,
            points: baselinePoints
          });
        }

        if (robustPoints.length) {
          series.push({
            name: `${method} � Robust`,
            points: robustPoints
          });
        }
      });

    return series;
  }

  // Compatibility fallback for older grouped API response shapes.
  const methods = data?.methods || data?.attacks || {};

  const entries = Array.isArray(methods)
    ? methods.map(method => [
        method.name || method.attack,
        method
      ])
    : Object.entries(methods);

  return entries
    .filter(([name]) => name && (!selected || name === selected))
    .flatMap(([name, payload]) => {
      const baseline = normalizeSeries(payload, "baseline_asr");
      const robust = normalizeSeries(payload, "robust_asr");

      const legacy = baseline.length
        ? baseline
        : normalizeSeries(payload, "attack_success_rate").length
          ? normalizeSeries(payload, "attack_success_rate")
          : normalizeSeries(payload, "asr");

      const result = [];

      if (baseline.length || legacy.length) {
        result.push({
          name: `${name} � Baseline`,
          points: baseline.length ? baseline : legacy
        });
      }

      if (robust.length) {
        result.push({
          name: `${name} � Robust`,
          points: robust
        });
      }

      return result;
    })
    .filter(series => series.points.length);
}

function heatColor(value, min, max) {
  if (!Number.isFinite(value)) {
    return "rgba(255,255,255,.03)";
  }

  const t = Math.max(
    0,
    Math.min(1, (value - min) / ((max - min) || 1))
  );

  if (value >= 0) {
    return `rgba(244,197,66,${0.12 + t * 0.72})`;
  }

  const a = Math.min(
    1,
    Math.abs(value - min) / (Math.abs(min) || 1)
  );

  return `rgba(32,201,216,${0.12 + a * 0.62})`;
}

function renderHeatmap(data) {
  els.heatmap.replaceChildren();

  const rows = Array.isArray(data) ? data : data?.data;

  if (!Array.isArray(rows) || !rows.length) {
    const empty = document.createElement("div");

    empty.className = "snr-empty";
    empty.textContent =
      "No generalization-gap data available from backend.";

    els.heatmap.appendChild(empty);
    return;
  }

  const attacks = [...new Set(rows.map(r => r.attack))];

  const snrs = [
    ...new Set(rows.map(r => Number(r.snr)))
  ].sort((a, b) => a - b);

  const values = rows
    .map(r => Number(r.gap))
    .filter(Number.isFinite);

  const min = Math.min(...values);
  const max = Math.max(...values);

  const grid = document.createElement("div");

  grid.className = "heatmap";

  grid.style.gridTemplateColumns =
    `110px repeat(${snrs.length}, minmax(25px, 1fr))`;

  grid.appendChild(document.createElement("div"));

  snrs.forEach(snr => {
    const cell = document.createElement("div");

    cell.className = "hm-label";
    cell.style.justifyContent = "center";
    cell.textContent = snr;

    grid.appendChild(cell);
  });

  attacks.forEach(attack => {
    const label = document.createElement("div");

    label.className = "hm-label";
    label.textContent = attack;

    grid.appendChild(label);

    snrs.forEach(snr => {
      const row = rows.find(
        r =>
          r.attack === attack &&
          Number(r.snr) === snr
      );

      const value = row ? Number(row.gap) : NaN;

      const cell = document.createElement("div");

      cell.className = "hm-cell";

      // Keep color scaling based on the original backend fraction.
      cell.style.background = heatColor(value, min, max);

      // Convert fraction to percentage points for display only.
      const percentagePoints = Number.isFinite(value)
        ? value * 100
        : NaN;

      cell.title =
        `${attack} � ${snr} dB � ${
          Number.isFinite(percentagePoints)
            ? percentagePoints.toFixed(2)
            : "N/A"
        } pp`;

      cell.textContent = Number.isFinite(percentagePoints)
        ? percentagePoints.toFixed(1)
        : "�";

      grid.appendChild(cell);
    });
  });

  els.heatmap.appendChild(grid);
}

          state.summary.user.name || "User";

        els.userRole.textContent =
          state.summary.user.role || "AMCShield Console";
      }
    }

    if (performance.status === "fulfilled") {
      state.performance = performance.value;
      renderPerformance(state.performance);
    } else {
      els.performanceLoading.textContent =
        "Performance data unavailable.";
    }

    if (attacks.status === "fulfilled") {
      state.attacks = attacks.value;
      renderAttackChart(state.attacks);
    } else {
      els.attackLoading.textContent =
        "Attack data unavailable.";
    }

    if (gap.status === "fulfilled") {
      state.gap = gap.value;
      renderHeatmap(state.gap);
    } else {
      renderHeatmap(null);
    }

    if (comparison.status === "fulfilled") {
      renderComparison(comparison.value);
    } else {
      renderComparison(null);
    }

    if (snr.status === "fulfilled") {
      state.snr = snr.value;
      populateSNR(state.snr);
    }

    if (updates.status === "fulfilled") {
      state.updates = updates.value;
      renderUpdates(state.updates);
    } else {
      renderUpdates(null);
    }

    const anySuccess = results.some(
      r => r.status === "fulfilled"
    );

    setAPIStatus(
      anySuccess,
      anySuccess
        ? "Backend data loaded"
        : "No endpoint responded"
    );

    if (!anySuccess) {
      showToast(
        "AMCShield backend is unavailable. No dummy data has been shown."
      );
    } else {
      setLastUpdated();
    }

  } catch (error) {
    console.error("Dashboard load failed:", error);

    setAPIStatus(false, "Connection error");

    showToast(
      "Unable to load AMCShield backend data."
    );

  } finally {
    setLoading(false);
  }
}

async function requireAuth() {
  try {
    const response = await fetch("/api/auth/me", {
      headers: {
        "Accept": "application/json"
      },
      credentials: "include"
    });

    if (!response.ok) {
      window.location.href = "/";
      return false;
    }

    const body = await response.json();
    const user = body.user || {};

    els.userName.textContent =
      user.name || user.login || "User";

    els.userRole.textContent =
      user.role || "AMCShield Console";

    return true;

  } catch (error) {
    console.error("Authentication check failed:", error);

    window.location.href = "/";
    return false;
  }
}

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

import pandas as pd


class ResultsService:
    """Read-only analytics service for existing AMCShield evaluation artifacts."""

    def __init__(self, results_dir: Path):
        self.results_dir = Path(results_dir)

    def exists(self) -> bool:
        return self.results_dir.is_dir()

    def list_files(self) -> list[str]:
        if not self.exists():
            return []
        return sorted(
            str(path.relative_to(self.results_dir))
            for path in self.results_dir.rglob("*")
            if path.is_file()
        )

    def summary(self) -> dict[str, Any]:
        return {
            "exists": self.exists(),
            "file_count": len(self.list_files()),
            "files": self.list_files(),
        }

    def read_json(self, filename: str) -> Any:
        path = self._safe_path(filename)
        return json.loads(path.read_text(encoding="utf-8"))

    def read_csv(self, filename: str) -> list[dict[str, Any]]:
        path = self._safe_path(filename)
        frame = pd.read_csv(path)
        return frame.where(pd.notna(frame), None).to_dict(orient="records")

    def _safe_path(self, filename: str) -> Path:
        root = self.results_dir.resolve()
        path = (root / filename).resolve()

        if root != path and root not in path.parents:
            raise ValueError("Invalid results path")

        if not path.is_file():
            raise FileNotFoundError(filename)

        return path

    def _load(self) -> tuple[dict[str, Any], pd.DataFrame, pd.DataFrame, pd.DataFrame]:
        metadata = self.read_json("metadata.json")
        clean = pd.DataFrame(self.read_csv("clean_per_snr.csv"))
        attacks = pd.DataFrame(self.read_csv("attack_results_per_snr.csv"))
        gaps = pd.DataFrame(self.read_csv("generalization_gap.csv"))

        required = {
            "clean": {"model", "snr_db", "n_samples", "n_correct", "accuracy"},
            "attacks": {
                "model", "attack", "snr_db", "n_samples",
                "n_originally_correct", "n_flipped",
                "accuracy_under_attack", "attack_success_rate",
            },
            "gaps": {
                "attack", "snr_db", "baseline_asr",
                "robust_asr", "gap_robust_minus_baseline",
            },
        }

        for name, frame in [
            ("clean", clean),
            ("attacks", attacks),
            ("gaps", gaps),
        ]:
            missing = required[name] - set(frame.columns)
            if missing:
                raise ValueError(
                    f"{name} CSV is missing required columns: {sorted(missing)}"
                )

        return metadata, clean, attacks, gaps

    @staticmethod
    def _records(frame: pd.DataFrame) -> list[dict[str, Any]]:
        return frame.where(pd.notna(frame), None).to_dict(orient="records")

    @staticmethod
    def _pooled_asr(frame: pd.DataFrame) -> float | None:
        denominator = int(frame["n_originally_correct"].sum())
        if denominator == 0:
            return None
        return float(frame["n_flipped"].sum() / denominator)

    def evaluation_summary(self) -> dict[str, Any]:
        metadata, clean, attacks, gaps = self._load()

        def clean_mean(model: str) -> float | None:
            rows = clean.loc[clean["model"] == model, "accuracy"]
            return float(rows.mean()) if not rows.empty else None

        def sample_total(model: str) -> int:
            rows = clean.loc[clean["model"] == model, "n_samples"]
            return int(rows.sum())

        baseline = attacks[attacks["model"] == "baseline"]
        robust = attacks[attacks["model"] == "robust"]

        baseline_asr = self._pooled_asr(baseline)
        robust_asr = self._pooled_asr(robust)

        all_originally_correct = int(attacks["n_originally_correct"].sum())
        all_flipped = int(attacks["n_flipped"].sum())
        mean_gap = (
            robust_asr - baseline_asr
            if baseline_asr is not None and robust_asr is not None
            else None
        )

        snrs = sorted(int(value) for value in clean["snr_db"].unique())

        return {
            "robust_clean_accuracy": clean_mean("robust"),
            "robust_samples": sample_total("robust"),
            "baseline_clean_accuracy": clean_mean("baseline"),
            "baseline_samples": sample_total("baseline"),
            "mean_robust_asr": robust_asr,
            "mean_baseline_asr": baseline_asr,
            "attack_scope": sorted(str(value) for value in attacks["attack"].unique()),
            "mean_gap": mean_gap,
            "gap_definition": "pooled robust ASR minus pooled baseline ASR",
            "snr_levels": snrs,
            "snr_min": min(snrs) if snrs else None,
            "snr_max": max(snrs) if snrs else None,
            "source": "evaluation CSV artifacts",
            "status": metadata.get("status", "unknown"),
            "message": "Metrics are calculated from the stored evaluation subset.",
            "test_samples": int(metadata.get("total_samples", 0)),
            "num_classes": int(metadata.get("num_classes", 0)),
            "dataset": metadata.get("dataset"),
            "attacks": sorted(str(value) for value in attacks["attack"].unique()),
            "total_attack_samples": int(attacks["n_samples"].sum()),
            "total_originally_correct": all_originally_correct,
            "total_flipped": all_flipped,
        }

    def evaluation_filters(self) -> dict[str, Any]:
        _, clean, attacks, _ = self._load()
        return {
            "models": sorted(str(value) for value in clean["model"].unique()),
            "attacks": sorted(str(value) for value in attacks["attack"].unique()),
            "snr_levels": sorted(int(value) for value in clean["snr_db"].unique()),
        }

    def evaluation_comparison(self) -> list[dict[str, Any]]:
        _, _, attacks, _ = self._load()
        rows = []

        for attack in sorted(attacks["attack"].unique()):
            baseline = attacks[
                (attacks["model"] == "baseline") &
                (attacks["attack"] == attack)
            ]
            robust = attacks[
                (attacks["model"] == "robust") &
                (attacks["attack"] == attack)
            ]

            rows.append({
                "attack": str(attack),
                "baseline_asr": self._pooled_asr(baseline),
                "robust_asr": self._pooled_asr(robust),
            })

        return rows

    def evaluation_attack_summary(self) -> list[dict[str, Any]]:
        _, _, attacks, gaps = self._load()
        rows = []

        for attack in sorted(attacks["attack"].unique()):
            baseline = attacks[
                (attacks["model"] == "baseline") &
                (attacks["attack"] == attack)
            ]
            robust = attacks[
                (attacks["model"] == "robust") &
                (attacks["attack"] == attack)
            ]

            baseline_asr = self._pooled_asr(baseline)
            robust_asr = self._pooled_asr(robust)
            gap_rows = gaps[gaps["attack"] == attack]

            rows.append({
                "attack": str(attack),
                "baseline_asr": baseline_asr,
                "robust_asr": robust_asr,
                "gap": (
                    robust_asr - baseline_asr
                    if baseline_asr is not None and robust_asr is not None
                    else None
                ),
                "generalization_gap": (
                    robust_asr - baseline_asr
                    if baseline_asr is not None and robust_asr is not None
                    else None
                ),
                "mean_per_snr_gap": (
                    float(gap_rows["gap_robust_minus_baseline"].mean())
                    if not gap_rows.empty else None
                ),
            })

        return rows

    def evaluation_results(
        self,
        attack: str | None = None,
        snr: int | None = None,
        model: str | None = None,
    ) -> list[dict[str, Any]]:
        _, clean, attacks, gaps = self._load()
        rows = []

        selected_attacks = (
            [attack] if attack and attack != "all"
            else sorted(attacks["attack"].unique())
        )

        for attack_name in selected_attacks:
            attack_rows = attacks[attacks["attack"] == attack_name]

            if snr is not None:
                attack_rows = attack_rows[attack_rows["snr_db"] == snr]

            for _, attack_row in attack_rows.iterrows():
                model_name = str(attack_row["model"])

                if model and model != "all" and model != model_name:
                    continue

                snr_value = int(attack_row["snr_db"])
                clean_row = clean[
                    (clean["model"] == model_name) &
                    (clean["snr_db"] == snr_value)
                ]

                baseline_gap = gaps[
                    (gaps["attack"] == attack_name) &
                    (gaps["snr_db"] == snr_value)
                ]

                accuracy = (
                    float(clean_row.iloc[0]["accuracy"])
                    if not clean_row.empty else None
                )

                gap_value = None
                if not baseline_gap.empty:
                    gap_value = float(
                        baseline_gap.iloc[0]["gap_robust_minus_baseline"]
                    )

                rows.append({
                    "model": model_name,
                    "attack": str(attack_name),
                    "snr": snr_value,
                    "samples": int(attack_row["n_samples"]),
                    "baseline_accuracy": (
                        accuracy if model_name == "baseline" else None
                    ),
                    "robust_accuracy": (
                        accuracy if model_name == "robust" else None
                    ),
                    "clean_accuracy": accuracy,
                    "accuracy_under_attack": float(
                        attack_row["accuracy_under_attack"]
                    ),
                    "attack_success_rate": float(
                        attack_row["attack_success_rate"]
                    ),
                    "baseline_asr": (
                        float(baseline_gap.iloc[0]["baseline_asr"])
                        if not baseline_gap.empty else None
                    ),
                    "robust_asr": (
                        float(baseline_gap.iloc[0]["robust_asr"])
                        if not baseline_gap.empty else None
                    ),
                    "gap": gap_value,
                    "mean_linf": float(attack_row["mean_linf"]),
                    "mean_l2": float(attack_row["mean_l2"]),
                })

        return rows

    def evaluation_accuracy_by_snr(self) -> list[dict[str, Any]]:
        _, clean, _, _ = self._load()
        pivot = clean.pivot(
            index="snr_db",
            columns="model",
            values="accuracy",
        ).reset_index()

        rows = []
        for _, row in pivot.iterrows():
            rows.append({
                "snr": int(row["snr_db"]),
                "baseline_accuracy": (
                    float(row["baseline"])
                    if "baseline" in pivot.columns and pd.notna(row.get("baseline"))
                    else None
                ),
                "robust_accuracy": (
                    float(row["robust"])
                    if "robust" in pivot.columns and pd.notna(row.get("robust"))
                    else None
                ),
            })

        return rows

    def evaluation_generalization_gap(self) -> list[dict[str, Any]]:
        _, _, _, gaps = self._load()
        result = gaps.groupby("snr_db", as_index=False)[
            "gap_robust_minus_baseline"
        ].mean()

        return [
            {"snr": int(row["snr_db"]), "gap": float(row["gap_robust_minus_baseline"])}
            for _, row in result.iterrows()
        ]

    def evaluation_insights(self) -> list[dict[str, Any]]:
        comparison = self.evaluation_comparison()
        clean_summary = self.evaluation_summary()
        findings = []

        for row in comparison:
            if row["baseline_asr"] is None or row["robust_asr"] is None:
                continue

            difference = row["robust_asr"] - row["baseline_asr"]
            direction = "lower" if difference < 0 else "higher"

            findings.append({
                "type": "comparison",
                "title": f"{row['attack'].upper()} pooled ASR comparison",
                "description": (
                    f"Robust model pooled ASR is {abs(difference) * 100:.2f} "
                    f"percentage points {direction} than baseline on this evaluation subset."
                ),
            })

        findings.append({
            "type": "clean_accuracy",
            "title": "Clean accuracy comparison",
            "description": (
                f"Mean clean accuracy across SNR levels: baseline "
                f"{clean_summary['baseline_clean_accuracy'] * 100:.2f}%, "
                f"robust {clean_summary['robust_clean_accuracy'] * 100:.2f}%."
            ),
        })

        return findings

    def dashboard_performance(self) -> list[dict[str, Any]]:
        return self.evaluation_accuracy_by_snr()

    def dashboard_attacks(self) -> list[dict[str, Any]]:
        _, _, attacks, _ = self._load()
        rows = []

        for attack in sorted(attacks["attack"].unique()):
            baseline = attacks[
                (attacks["model"] == "baseline") &
                (attacks["attack"] == attack)
            ]
            robust = attacks[
                (attacks["model"] == "robust") &
                (attacks["attack"] == attack)
            ]

            for _, row in baseline.iterrows():
                snr_value = int(row["snr_db"])
                robust_row = robust[robust["snr_db"] == snr_value]

                rows.append({
                    "attack": str(attack),
                    "snr": snr_value,
                    "baseline_asr": float(row["attack_success_rate"]),
                    "robust_asr": (
                        float(robust_row.iloc[0]["attack_success_rate"])
                        if not robust_row.empty else None
                    ),
                })

        return rows

    def dashboard_generalization_gap(self) -> list[dict[str, Any]]:
        _, _, _, gaps = self._load()
        return [
            {
                "attack": str(row["attack"]),
                "snr": int(row["snr_db"]),
                "gap": float(row["gap_robust_minus_baseline"]),
            }
            for _, row in gaps.iterrows()
        ]

    def dashboard_snr(self) -> list[dict[str, Any]]:
        _, clean, attacks, _ = self._load()
        rows = []

        for snr_value in sorted(clean["snr_db"].unique()):
            baseline = clean[
                (clean["model"] == "baseline") &
                (clean["snr_db"] == snr_value)
            ]
            robust = clean[
                (clean["model"] == "robust") &
                (clean["snr_db"] == snr_value)
            ]

            row: dict[str, Any] = {
                "snr": int(snr_value),
                "baseline_accuracy": (
                    float(baseline.iloc[0]["accuracy"]) if not baseline.empty else None
                ),
                "robust_accuracy": (
                    float(robust.iloc[0]["accuracy"]) if not robust.empty else None
                ),
            }

            for attack_name in ["fgsm", "pgd", "mim", "cw", "blackbox"]:
                for model_name, prefix in [
                    ("baseline", "baseline"),
                    ("robust", "robust"),
                ]:
                    match = attacks[
                        (attacks["model"] == model_name) &
                        (attacks["attack"] == attack_name) &
                        (attacks["snr_db"] == snr_value)
                    ]
                    row[f"{prefix}_{attack_name}_asr"] = (
                        float(match.iloc[0]["attack_success_rate"])
                        if not match.empty else None
                    )

                robust_match = attacks[
                    (attacks["model"] == "robust") &
                    (attacks["attack"] == attack_name) &
                    (attacks["snr_db"] == snr_value)
                ]
                row[f"{attack_name}_asr"] = (
                    float(robust_match.iloc[0]["attack_success_rate"])
                    if not robust_match.empty else None
                )

            rows.append(row)

        return rows

    def dashboard_summary(self) -> dict[str, Any]:
        metadata, clean, attacks, _ = self._load()
        snrs = sorted(int(value) for value in clean["snr_db"].unique())

        return {
            "dataset": {
                "name": metadata.get("dataset"),
                "num_classes": int(metadata.get("num_classes", 0)),
                "snr_min": min(snrs) if snrs else None,
                "snr_max": max(snrs) if snrs else None,
            },
            "evaluation": {
                "test_samples": int(metadata.get("total_samples", 0)),
                "snr_levels": len(snrs),
                "attack_methods": sorted(
                    str(value) for value in attacks["attack"].unique()
                ),
            },
            "attacks": sorted(
                str(value) for value in attacks["attack"].unique()
            ),
        }

    def dashboard_recent_updates(self) -> list[dict[str, Any]]:
        # Evaluation metadata does not contain dated activity records.
        return []

/*
 * AMCShield Dataset page.
 * No experiment values are hardcoded here.
 * All metadata, distributions, samples and quality checks come from FastAPI.

const API_BASE_URL = "http://localhost:8000/api";

const $ = id => document.getElementById(id);
const state = { meta:null, classes:null, snr:null, quality:null };

async function api(path, options={}) {
  const r = await fetch(API_BASE_URL + path, {
    ...options,
    headers: {Accept:"application/json", ...(options.headers||{})}
  });
  if (!r.ok) {
    let detail="";
    try { const x=await r.json(); detail=x.detail||x.message||""; } catch(_){}
    throw new Error(detail || `HTTP ${r.status}`);
  }
  return r.json();
}

function setAPI(ok, detail) {
  const box=$("apiStatus");
  box.classList.toggle("online",ok);
  box.classList.toggle("offline",!ok);
  $("apiText").textContent=ok?"API ONLINE":"API OFFLINE";
  $("apiDetail").textContent=detail || (ok?"Dataset backend connected":"Backend unavailable");
}

function fmt(n) {
  if(n===null||n===undefined||!Number.isFinite(Number(n))) return "—";
  return new Intl.NumberFormat("en-IN",{maximumFractionDigits:0}).format(Number(n));
}
function pct(n) {
  if(n===null||n===undefined||!Number.isFinite(Number(n))) return "—";
  return `${Number(n).toFixed(1)}%`;
}

function renderMeta(d) {
  state.meta=d;
  const x=d.dataset||d;
  $("datasetName").textContent=x.name ?? "—";
  $("datasetPath").textContent=x.path ?? x.source ?? "Backend metadata";
  $("totalSamples").textContent=fmt(x.total_samples ?? x.samples);
  $("sampleSplit").textContent=x.test_samples!==undefined ? `Test: ${fmt(x.test_samples)}` : "Dataset total";
  $("classCount").textContent=fmt(x.num_classes ?? x.class_count);
  $("classDetail").textContent=x.classes_path ?? "Class metadata";
  $("signalLength").textContent=fmt(x.signal_length);
  $("inputShape").textContent=x.input_shape ? String(x.input_shape) : "Samples × channels × length";
  $("snrLevels").textContent=fmt(x.snr_levels ?? (Array.isArray(x.snr_values)?x.snr_values.length:null));
  const lo=x.snr_min, hi=x.snr_max;
  $("snrRange").textContent=Number.isFinite(Number(lo))&&Number.isFinite(Number(hi))?`${lo} dB to ${hi} dB`:"Backend SNR range";
  $("channels").textContent=fmt(x.input_channels ?? x.channels);
  $("channelDetail").textContent=x.channel_names ? x.channel_names.join(", ") : "Signal channels";
  $("sourceName").textContent=x.name || "Backend";
  $("datasetStatus").textContent="DATASET READY";
  $("datasetStatusDetail").textContent="Metadata loaded from backend";
  $("formatBadge").textContent=x.format || "Backend";
  const specs=[
    ["Name",x.name],["Source",x.source],["Path",x.path],["Format",x.format],
    ["Classes",x.num_classes ?? x.class_count],["Signal length",x.signal_length],
    ["Input channels",x.input_channels ?? x.channels],["SNR range",x.snr_min!==undefined?`${x.snr_min} to ${x.snr_max} dB`:null],
    ["SNR step",x.snr_step!==undefined?`${x.snr_step} dB`:null],["Train split",x.train_size],["Validation split",x.val_size],["Test split",x.test_size]
  ];
  const grid=$("specGrid"); grid.innerHTML="";
  specs.forEach(([k,v])=>{
    if(v===undefined||v===null) return;
    const el=document.createElement("div"); el.className="spec";
    el.innerHTML=`<span>${escapeHTML(k)}</span><strong>${escapeHTML(String(v))}</strong>`;
    grid.appendChild(el);
  });
}

function renderClassDistribution(data) {
  const rows=Array.isArray(data)?data:(data.data||data.classes||[]);
  const normalized=rows.map(x=>({
    name:x.name??x.class_name??x.label??x.modulation,
    value:Number(x.samples??x.count??x.total??x.value)
  })).filter(x=>x.name!=null&&Number.isFinite(x.value));
  const box=$("classChart"); box.innerHTML="";
  if(!normalized.length){box.innerHTML='<div class="empty">No class distribution returned by backend.</div>';return;}
  const max=Math.max(...normalized.map(x=>x.value));
  normalized.sort((a,b)=>b.value-a.value);
  normalized.forEach(x=>{
    const row=document.createElement("div");row.className="bar-row";
    row.innerHTML=`<span class="bar-label" title="${escapeHTML(String(x.name))}">${escapeHTML(String(x.name))}</span><div class="bar-track"><div class="bar-fill" style="width:${max?x.value/max*100:0}%"></div></div><span class="bar-value">${fmt(x.value)}</span>`;
    box.appendChild(row);
  });
  populateClassSelect(normalized);
}

function renderSNRDistribution(data) {
  const rows=Array.isArray(data)?data:(data.data||data.snr||[]);
  const normalized=rows.map(x=>({
    snr:Number(x.snr??x.level??x.value),
    value:Number(x.samples??x.count??x.total??x.value_count??x.frequency)
  })).filter(x=>Number.isFinite(x.snr)&&Number.isFinite(x.value)).sort((a,b)=>a.snr-b.snr);
  const box=$("snrChart");box.innerHTML="";
  if(!normalized.length){box.innerHTML='<div class="empty">No SNR distribution returned by backend.</div>';return;}
  const max=Math.max(...normalized.map(x=>x.value));
  const chart=document.createElement("div");chart.className="snr-grid";
  normalized.forEach(x=>{
    const col=document.createElement("div");col.className="snr-col";
    const bar=document.createElement("div");bar.className="snr-bar";bar.style.height=`${max?x.value/max*100:0}%`;bar.title=`${x.snr} dB · ${fmt(x.value)} samples`;
    const val=document.createElement("span");val.className="snr-value";val.textContent=fmt(x.value);
    const lab=document.createElement("span");lab.className="snr-label";lab.textContent=x.snr;
    col.append(bar,val,lab);chart.appendChild(col);
  });
  box.appendChild(chart);
  populateSNRSelect(normalized.map(x=>x.snr));
}

}

function renderSplits(d) {
  const x=d.splits||d;
  const items=[
    ["Train",x.train??x.train_size??x.training],
    ["Validation",x.validation??x.val??x.val_size],
    ["Test",x.test??x.test_size]
  ].filter(x=>x[1]!==undefined&&x[1]!==null);
  const box=$("splitList");box.innerHTML="";
  if(!items.length){box.innerHTML='<div class="empty">No split information returned by backend.</div>';return;}
  items.forEach(([name,value])=>{
    let p=typeof value==="number"?value:(Number(value)||0);
    if(p>1) p=p/100;
    const row=document.createElement("div");row.className="split";
    row.innerHTML=`<label>${escapeHTML(name)}</label><div class="split-track"><div class="split-fill" style="width:${Math.max(0,Math.min(100,p*100))}%"></div></div><span class="split-value">${typeof value==="number"&&value<=1?pct(value):String(value)}</span>`;
    box.appendChild(row);
  });
}

function renderQuality(d) {
  const rows=Array.isArray(d)?d:(d.checks||d.data||[]);
  const box=$("qualityGrid");box.innerHTML="";
  if(!rows.length){box.innerHTML='<div class="empty">No integrity checks returned by backend.</div>';return;}
  rows.forEach(x=>{
    const ok=x.ok??x.passed??x.status==="passed";
    const card=document.createElement("div");card.className="quality";
    card.innerHTML=`<div class="quality-top"><strong>${escapeHTML(String(x.name||x.check||"Check"))}</strong><span class="${ok?'ok':''}">${ok?'PASSED':String(x.status||'REVIEW')}</span></div><small>${escapeHTML(String(x.detail??x.message??"Backend result"))}</small>`;
    box.appendChild(card);
  });
}

function escapeHTML(s){return s.replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[c]));}

function renderSample(d) {
  const values=Array.isArray(d)?d:(d.samples||d.values||d.waveform||[]);
  const points=values.map(v=>Array.isArray(v)?Number(v[0]):Number(v)).filter(Number.isFinite);
  const box=$("sampleArea");
  if(!points.length){box.innerHTML='<div class="sample-placeholder"><strong>No waveform returned</strong><span>The backend returned no numeric sample for this selection.</span></div>';return;}
  const max=Math.max(...points.map(Math.abs))||1;
  const W=1000,H=210,mid=105;
  const step=W/Math.max(1,points.length-1);
  const dpath=points.map((v,i)=>`${i?'L':'M'} ${(i*step).toFixed(2)} ${(mid-(v/max)*85).toFixed(2)}`).join(" ");
  box.innerHTML=`<svg class="wave" viewBox="0 0 ${W} ${H}" preserveAspectRatio="none"><path class="grid" d="M0 30H1000M0 105H1000M0 180H1000"></path><path d="${dpath}"></path></svg>`;
}

async function loadSample(){
  const cls=$("classSelect").value,snr=$("snrSelect").value;
  if(!cls||snr===""){alert("Select both a modulation class and SNR level.");return;}
  $("sampleArea").innerHTML='<div class="sample-placeholder"><strong>Loading waveform…</strong><span>Fetching the selected signal from the backend.</span></div>';
  try{
    const q=`?class_name=${encodeURIComponent(cls)}&snr=${encodeURIComponent(snr)}`;
    renderSample(await api("/dataset/sample"+q));
  }catch(e){
    $("sampleArea").innerHTML='<div class="sample-placeholder"><strong>Sample unavailable</strong><span>The backend did not return a waveform for this selection.</span></div>';
  }
}

async function load(){
  $("refresh").disabled=true;
  try{
    const results=await Promise.allSettled([
      api("/dataset/metadata"),
      api("/dataset/class-distribution"),
      api("/dataset/snr-distribution"),
      api("/dataset/splits"),
      api("/dataset/quality")
    ]);
    let any=false;
    if(results[0].status==="fulfilled"){renderMeta(results[0].value);any=true}
    if(results[1].status==="fulfilled"){state.classes=results[1].value;renderClassDistribution(state.classes);any=true}
    if(results[2].status==="fulfilled"){state.snr=results[2].value;renderSNRDistribution(state.snr);any=true}
    if(results[3].status==="fulfilled"){renderSplits(results[3].value);any=true}
    if(results[4].status==="fulfilled"){state.quality=results[4].value;renderQuality(state.quality);any=true}
    setAPI(any,"Dataset endpoints "+(any?"responding":"unavailable"));
    $("lastUpdated").textContent=any?`Loaded ${new Date().toLocaleTimeString([], {hour:"2-digit",minute:"2-digit",second:"2-digit"})}`:"Not loaded";
  }catch(e){setAPI(false,"Connection error")}
  finally{$("refresh").disabled=false}
}

from pathlib import Path
from typing import Any

class DatasetService:
    def __init__(self, data_dir: Path):
        self.data_dir = Path(data_dir)

    def overview(self) -> dict[str, Any]:
        files = []
        if self.data_dir.exists():
            files = sorted(
                str(p.relative_to(self.data_dir))
                for p in self.data_dir.rglob("*")
                if p.is_file()
            )
        return {
            "exists": self.data_dir.exists(),
            "files": files,
        }

from fastapi import Depends, HTTPException, Request
from sqlalchemy.orm import Session

from .database.db import get_db
from .database.models import User, UserRole, UserStatus


def _db_user_from_session(request: Request, db: Session) -> User | None:
    session_user = request.session.get("user")
    if not session_user:
        return None

    if session_user.get("id") == "local-admin":
        return None

    provider = session_user.get("provider")
    user_id = session_user.get("id")

    if provider not in {"local", "github"} or not isinstance(user_id, int):
        return None

    user = db.get(User, user_id)
    if not user or user.status != UserStatus.ACTIVE.value:
        return None

    return user


def is_admin_role(user: dict | None) -> bool:
    if not user:
        return False
    role = str(
        user.get("role_key") or user.get("role") or ""
    ).strip().lower()
    return role in {"admin", "administrator"}


def is_admin_session(request: Request, db: Session) -> bool:
    session_user = request.session.get("user")
    if not session_user or not is_admin_role(session_user):
        return False

    if session_user.get("id") == "local-admin":
        return True

    if session_user.get("provider") != "local" or not isinstance(
        session_user.get("id"), int
    ):
        return False

    user = db.get(User, session_user["id"])
    return bool(
        user
        and user.status == UserStatus.ACTIVE.value
        and user.role == UserRole.ADMIN.value
    )


def require_authenticated(
    request: Request,
    db: Session = Depends(get_db),
) -> dict:
    session_user = request.session.get("user")
    if not session_user:
        raise HTTPException(
            status_code=401,
            detail="Authentication required.",
        )

    if (
        session_user.get("provider") == "local"
        and session_user.get("id") == "local-admin"
    ):
        return session_user

    db_user = _db_user_from_session(request, db)
    if not db_user:
        # Remove a stale/disabled session so it cannot be reused.
        request.session.clear()
        raise HTTPException(
            status_code=401,
            detail="Authentication required.",
        )

    # Rebuild authorization fields from the database on every protected request.
    # This prevents stale/tampered session role data from granting privileges.
    session_user["id"] = db_user.id
    session_user["name"] = db_user.name
    session_user["email"] = db_user.email
    session_user["role_key"] = db_user.role
    session_user["role"] = (
        "Administrator"
        if db_user.role == UserRole.ADMIN.value
        else "User"
    )
    return session_user

@router.get("/github/callback")
async def github_callback(
    request: Request,
    code: str | None = None,
    state: str | None = None,
    db: Session = Depends(get_db),
):
    """Complete GitHub OAuth only for an existing active AMCShield account."""
    expected_state = request.session.pop("github_oauth_state", None)
    state_created_at = request.session.pop(
        "github_oauth_state_created_at",
        None,
    )

    state_is_fresh = False
    if isinstance(state_created_at, int):
        state_is_fresh = (
            0 <= time() - state_created_at
            <= settings.github_state_max_age
        )

    if (
        not code
        or not state
        or not expected_state
        or not state_is_fresh
        or not secrets.compare_digest(state, expected_state)
    ):
        return RedirectResponse(
            url="/?auth_error=invalid_oauth_state",
            status_code=302,
        )

    if not settings.github_client_id or not settings.github_client_secret:
        return RedirectResponse(
            url="/?auth_error=github_not_configured",
            status_code=302,
        )

    try:
        async with httpx.AsyncClient(
            timeout=15.0,
            follow_redirects=False,
        ) as client:
            token_response = await client.post(
                GITHUB_TOKEN_URL,
                data={
                    "client_id": settings.github_client_id,
                    "client_secret": settings.github_client_secret,
                    "code": code,
                    "redirect_uri": settings.github_redirect_uri,
                },
                headers={
                    "Accept": "application/json",
                    "User-Agent": "AMCShield",
                },
            )
            token_response.raise_for_status()
            token_data = token_response.json()
            access_token = token_data.get("access_token")
            if not access_token:
                raise RuntimeError("GitHub did not return an access token.")

            github_headers = {
                "Accept": "application/vnd.github+json",
                "Authorization": f"Bearer {access_token}",
                "X-GitHub-Api-Version": "2022-11-28",
                "User-Agent": "AMCShield",
            }

            user_response = await client.get(
                GITHUB_USER_URL,
                headers=github_headers,
            )
            user_response.raise_for_status()
            github_user = user_response.json()

            # Always consult GitHub's email endpoint and require a verified
            # email. The public profile email alone is not sufficient.
            email_response = await client.get(
                GITHUB_EMAILS_URL,
                headers=github_headers,
            )
            email_response.raise_for_status()
            emails = email_response.json()

            verified_emails = [
                item
                for item in emails
                if item.get("verified") and item.get("email")
            ]
            primary_verified = next(
                (
                    item
                    for item in verified_emails
                    if item.get("primary")
                ),
                None,
            )
            selected_email = (
                primary_verified or
                (verified_emails[0] if verified_emails else None)
            )

        if not selected_email:
            return RedirectResponse(
                url="/?auth_error=github_email_unverified",
                status_code=302,
            )

        email = str(selected_email["email"]).strip().lower()
        db_user = db.scalar(
            select(User).where(User.email == email)
        )

        # GitHub authentication does not itself authorize AMCShield access.
        if not db_user:
            return RedirectResponse(
                url="/?auth_error=github_account_not_authorized",
                status_code=302,
            )

        if db_user.status != UserStatus.ACTIVE.value:
            return RedirectResponse(
                url="/?auth_error=github_account_not_active",
                status_code=302,
            )

        db_user.last_login_at = utcnow()
        db.commit()

        session_user = {
            "provider": "github",
            "id": db_user.id,
            "github_id": github_user.get("id"),
            "login": github_user.get("login"),
            "name": db_user.name,
            "email": db_user.email,
            "avatar_url": github_user.get("avatar_url"),
            # These values are derived from the DB record, not GitHub/client input.
            "role": (
                "Administrator"
                if db_user.role == UserRole.ADMIN.value
                else "User"
            ),
            "role_key": db_user.role,
        }

        # Replace any pre-authentication session state to prevent fixation.
        _start_session(request, session_user)

        return RedirectResponse(
            url="/dashboard.html",
            status_code=302,
        )

    except Exception as exc:
        # Do not expose provider secrets or raw upstream responses.
        print(f"GitHub OAuth error: {type(exc).__name__}")
        return RedirectResponse(
            url="/?auth_error=github_auth_failed",
            status_code=302,
        )

def login(
    payload: LoginRequest,
    request: Request,
    db: Session = Depends(get_db),
):
    """
    Login for:
    1. Bootstrap administrator from .env
    2. Activated database users
    """

    identity = payload.identity.strip().lower()
    client_ip = request.client.host if request.client else None

    # Check the same rate-limit bucket before doing password work.
    retry_after = check_login_rate_limit(db, identity, client_ip)
    if retry_after:
        raise HTTPException(
            status_code=429,
            detail="Too many failed login attempts. Please try again later.",
            headers={"Retry-After": str(retry_after)},
        )

    # ---------------------------------------------------------
    # 1. Bootstrap administrator
    # ---------------------------------------------------------

    if _bootstrap_admin_matches(identity, payload.password):
        clear_login_failures(db, identity, client_ip)

        session_user = {
            "provider": "local",
            "id": "local-admin",
            "login": settings.admin_username,
            "name": settings.admin_username,
            "email": settings.admin_email,
            "avatar_url": None,
            "role": "Administrator",
            "role_key": "admin",
        }

        _start_session(request, session_user)

        return {
            "success": True,
            "user": session_user,
        }

    # ---------------------------------------------------------
    # 2. Database-backed invited user
    # ---------------------------------------------------------

    user = db.scalar(
        select(User).where(User.email == identity)
    )

    # Do a dummy Argon2 verification so nonexistent-user
    # responses don't have an obvious timing difference.

    if user is None:
        verify_password(
            _DUMMY_HASH,
            payload.password,
        )
        retry_after = record_login_failure(db, identity, client_ip)
        headers = {"Retry-After": str(retry_after)} if retry_after else None
        raise HTTPException(
            status_code=429 if retry_after else 401,
            detail=(
                "Too many failed login attempts. Please try again later."
                if retry_after
                else "Invalid username/email or password."
            ),
            headers=headers,
        )

    # ---------------------------------------------------------
    # 3. Pending invitation
    # ---------------------------------------------------------

    if user.status != UserStatus.ACTIVE.value:
        retry_after = record_login_failure(db, identity, client_ip)
        headers = {"Retry-After": str(retry_after)} if retry_after else None
        raise HTTPException(
            status_code=429 if retry_after else 401,
            detail=(
                "Too many failed login attempts. Please try again later."
                if retry_after
                else "Invalid username/email or password."
            ),
            headers=headers,
        )

    # ---------------------------------------------------------
    # 5. Verify Argon2 password
    # ---------------------------------------------------------

    if (
        not user.password_hash
        or not verify_password(
            user.password_hash,
            payload.password,
        )
    ):
        retry_after = record_login_failure(db, identity, client_ip)
        headers = {"Retry-After": str(retry_after)} if retry_after else None
        raise HTTPException(
            status_code=429 if retry_after else 401,
            detail=(
                "Too many failed login attempts. Please try again later."
                if retry_after
                else "Invalid username/email or password."
            ),
            headers=headers,
        )

    # ---------------------------------------------------------
    # 6. Update last login
    # ---------------------------------------------------------

    user.last_login_at = utcnow()
    db.commit()
    clear_login_failures(db, identity, client_ip)

    # ---------------------------------------------------------
    # 7. Create session
    # ---------------------------------------------------------

    session_user = {
        "provider": "local",
        "id": user.id,
        "login": user.email,
        "name": user.name,
        "email": user.email,
        "avatar_url": None,
        "role": _ROLE_LABELS.get(
            user.role,
            "User",
        ),
        "role_key": user.role,
    }

    _start_session(
        request,
        session_user,
    )

    return {
        "success": True,
        "user": session_user,
    }

/*
 * AMCShield frontend security helper.
 *
 * All authenticated state-changing requests should use amcFetch().
 * Login remains intentionally exempt because the backend login route
 * does not require a CSRF token.
 */
(function () {
  let csrfToken = null;
  let csrfPromise = null;

  async function fetchCsrfToken(forceRefresh = false) {
    if (csrfToken && !forceRefresh) {
      return csrfToken;
    }

    if (csrfPromise && !forceRefresh) {
      return csrfPromise;
    }

    csrfPromise = fetch("/api/auth/csrf", {
      method: "GET",
      credentials: "include",
      headers: {
        "Accept": "application/json"
      },
      cache: "no-store"
    })
      .then(async (response) => {
        if (!response.ok) {
          throw new Error(`CSRF token request failed (${response.status})`);
        }

        const body = await response.json();

        if (!body || typeof body.csrf_token !== "string" || !body.csrf_token) {
          throw new Error("CSRF token was not returned by the server.");
        }

        csrfToken = body.csrf_token;
        return csrfToken;
      })
      .finally(() => {
        csrfPromise = null;
      });

    return csrfPromise;
  }

  async function amcFetch(url, options = {}) {
    const requestOptions = { ...options };
    const method = String(requestOptions.method || "GET").toUpperCase();

    requestOptions.credentials = requestOptions.credentials || "include";
    requestOptions.headers = new Headers(requestOptions.headers || {});

    requestOptions.headers.set("Accept", "application/json");

    const stateChanging =
      !["GET", "HEAD", "OPTIONS"].includes(method);

    if (stateChanging) {
      let token = await fetchCsrfToken();

      requestOptions.headers.set("X-CSRF-Token", token);

      let response = await fetch(url, requestOptions);

      // If the session/token changed, refresh once and retry.
      if (response.status === 403) {
        try {
          token = await fetchCsrfToken(true);
          requestOptions.headers.set("X-CSRF-Token", token);
          response = await fetch(url, requestOptions);
        } catch (_) {
          // Preserve the original 403 when refresh itself fails.
        }
      }

      return response;
    }

    return fetch(url, requestOptions);
  }

  window.AMCShieldSecurity = Object.freeze({
    getCsrfToken: fetchCsrfToken,
    fetch: amcFetch
  });
})();
